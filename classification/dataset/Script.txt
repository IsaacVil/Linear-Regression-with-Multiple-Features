-- =============================================

-- =============================================
-- TABLAS DE USUARIOS Y AUTENTICACIÓN
-- =============================================

CREATE TABLE Users (
    UserId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    Email NVARCHAR(255) NOT NULL UNIQUE,
    PhoneNumber NVARCHAR(20) NOT NULL,
    PasswordHash NVARCHAR(255) NOT NULL,
    FirstName NVARCHAR(100) NOT NULL,
    LastName NVARCHAR(100) NOT NULL,
    ProfileImageUrl NVARCHAR(500),
    IsActive BIT DEFAULT 1,
    AcceptedMarketingConsent BIT DEFAULT 0,
    DateOfBirth DATE,
    CreatedAt DATETIME2 DEFAULT GETUTCDATE(),
    UpdatedAt DATETIME2 DEFAULT GETUTCDATE(),
    LastLoginAt DATETIME2,
    INDEX IX_Users_Email (Email),
    INDEX IX_Users_PhoneNumber (PhoneNumber)
);

-- =============================================
-- TABLAS DE NEGOCIOS (MULTI-TENANCY)
-- =============================================

    Name NVARCHAR(100) NOT NULL UNIQUE,

    PricePerAppointment DECIMAL(10,2) NOT NULL, -- Precio por cita adicional
    IconUrl NVARCHAR(500),
    IsActive BIT DEFAULT 1,
    CreatedAt DATETIME2 DEFAULT GETUTCDATE()
);

CREATE TABLE Businesses (
    BusinessId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    OwnerId UNIQUEIDENTIFIER NOT NULL,
    IndustryId INT NOT NULL,
    BusinessName NVARCHAR(200) NOT NULL,
    LegalName NVARCHAR(200),
    TaxId NVARCHAR(50), -- Cédula jurídica CR
    Description NVARCHAR(1000),
    LogoUrl NVARCHAR(500),
    CoverImageUrl NVARCHAR(500),
    PrimaryColor NVARCHAR(7) DEFAULT '#000000', -- HEX color
    SecondaryColor NVARCHAR(7) DEFAULT '#FFFFFF',
    Phone NVARCHAR(20),
    Email NVARCHAR(255),
    WebsiteUrl NVARCHAR(500),
    FacebookUrl NVARCHAR(500),
    InstagramUrl NVARCHAR(500),
    WhatsAppNumber NVARCHAR(20),
    GoogleMyBusinessUrl NVARCHAR(500),
    IsActive BIT DEFAULT 1,
    SubscriptionStatus NVARCHAR(20) DEFAULT 'trial', -- trial, active, suspended, cancelled
    SubscriptionStartDate DATETIME2,
    SubscriptionEndDate DATETIME2,
    TrialEndsAt DATETIME2, -- Primer mes gratis
    LastPaymentDate DATETIME2,
    CreatedAt DATETIME2 DEFAULT GETUTCDATE(),
    UpdatedAt DATETIME2 DEFAULT GETUTCDATE(),
    FOREIGN KEY (OwnerId) REFERENCES Users(UserId),
    FOREIGN KEY (IndustryId) REFERENCES Industries(IndustryId),
    INDEX IX_Businesses_OwnerId (OwnerId),
    INDEX IX_Businesses_SubscriptionStatus (SubscriptionStatus)
);

CREATE TABLE BusinessLocations (
    LocationId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    BusinessId UNIQUEIDENTIFIER NOT NULL,
    LocationName NVARCHAR(200) NOT NULL,
    Address NVARCHAR(500) NOT NULL,
    City NVARCHAR(100) NOT NULL,
    State NVARCHAR(100) NOT NULL, -- Provincia en CR
    Country NVARCHAR(100) DEFAULT 'Costa Rica',
    PostalCode NVARCHAR(20),
    Latitude DECIMAL(10,8),
    Longitude DECIMAL(11,8),
    Phone NVARCHAR(20),
    Email NVARCHAR(255),
    IsMainLocation BIT DEFAULT 0,
    IsActive BIT DEFAULT 1,
    CreatedAt DATETIME2 DEFAULT GETUTCDATE(),
    UpdatedAt DATETIME2 DEFAULT GETUTCDATE(),
    FOREIGN KEY (BusinessId) REFERENCES Businesses(BusinessId) ON DELETE CASCADE,
    INDEX IX_BusinessLocations_BusinessId (BusinessId)
);

CREATE TABLE BusinessHours (
    HourId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    LocationId UNIQUEIDENTIFIER NOT NULL,
    DayOfWeek INT NOT NULL, -- 0=Domingo, 6=Sábado
    OpenTime TIME NOT NULL,
    CloseTime TIME NOT NULL,
    IsOpen BIT DEFAULT 1,
    FOREIGN KEY (LocationId) REFERENCES BusinessLocations(LocationId) ON DELETE CASCADE,
    INDEX IX_BusinessHours_LocationId (LocationId)
);

CREATE TABLE BusinessClosedDates (
    ClosedDateId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    LocationId UNIQUEIDENTIFIER NOT NULL,
    ClosedDate DATE NOT NULL,
    Reason NVARCHAR(200),
    CreatedAt DATETIME2 DEFAULT GETUTCDATE(),
    FOREIGN KEY (LocationId) REFERENCES BusinessLocations(LocationId) ON DELETE CASCADE,
    INDEX IX_BusinessClosedDates_LocationId_Date (LocationId, ClosedDate)
);

CREATE TABLE BusinessEmployees (
    EmployeeId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    BusinessId UNIQUEIDENTIFIER NOT NULL,
    LocationId UNIQUEIDENTIFIER, -- NULL = disponible en todas las ubicaciones
    UserId UNIQUEIDENTIFIER NOT NULL,
    RoleId INT NOT NULL,
    DisplayName NVARCHAR(200), -- Nombre profesional
    Title NVARCHAR(100), -- Título/Especialidad
    Bio NVARCHAR(1000),
    ProfileImageUrl NVARCHAR(500),
    IsActive BIT DEFAULT 1,
    GoogleCalendarId NVARCHAR(255), -- Para sincronización
    HireDate DATE,
    CreatedAt DATETIME2 DEFAULT GETUTCDATE(),
    UpdatedAt DATETIME2 DEFAULT GETUTCDATE(),
    FOREIGN KEY (BusinessId) REFERENCES Businesses(BusinessId) ON DELETE CASCADE,
    FOREIGN KEY (LocationId) REFERENCES BusinessLocations(LocationId),
    FOREIGN KEY (UserId) REFERENCES Users(UserId),
    FOREIGN KEY (RoleId) REFERENCES BusinessRoles(RoleId),
    INDEX IX_BusinessEmployees_BusinessId (BusinessId),
    INDEX IX_BusinessEmployees_UserId (UserId),
    UNIQUE (BusinessId, UserId) -- Un usuario solo puede tener un rol por negocio
);

CREATE TABLE EmployeeWorkingHours (
    WorkingHourId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    EmployeeId UNIQUEIDENTIFIER NOT NULL,
    DayOfWeek INT NOT NULL,
    StartTime TIME NOT NULL,
    EndTime TIME NOT NULL,
    IsAvailable BIT DEFAULT 1,
    FOREIGN KEY (EmployeeId) REFERENCES BusinessEmployees(EmployeeId) ON DELETE CASCADE,
    INDEX IX_EmployeeWorkingHours_EmployeeId (EmployeeId)
);

CREATE TABLE EmployeeTimeOff (
    TimeOffId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    EmployeeId UNIQUEIDENTIFIER NOT NULL,
    StartDate DATETIME2 NOT NULL,
    EndDate DATETIME2 NOT NULL,
    Reason NVARCHAR(200),
    CreatedAt DATETIME2 DEFAULT GETUTCDATE(),
    FOREIGN KEY (EmployeeId) REFERENCES BusinessEmployees(EmployeeId) ON DELETE CASCADE,
    INDEX IX_EmployeeTimeOff_EmployeeId (EmployeeId)
);

-- =============================================
-- TABLAS DE SERVICIOS
-- =============================================

CREATE TABLE ServiceCategories (
    CategoryId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    BusinessId UNIQUEIDENTIFIER NOT NULL,
    CategoryName NVARCHAR(100) NOT NULL,
    DisplayOrder INT DEFAULT 0,
    IsActive BIT DEFAULT 1,
    FOREIGN KEY (BusinessId) REFERENCES Businesses(BusinessId) ON DELETE CASCADE,
    INDEX IX_ServiceCategories_BusinessId (BusinessId)
);

CREATE TABLE Services (
    ServiceId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    BusinessId UNIQUEIDENTIFIER NOT NULL,
    CategoryId UNIQUEIDENTIFIER,
    ServiceName NVARCHAR(200) NOT NULL,
    Description NVARCHAR(1000),
    DurationMinutes INT NOT NULL,
    Price DECIMAL(10,2) NOT NULL,
    ImageUrl NVARCHAR(500),
    IsActive BIT DEFAULT 1,
    RequiresDeposit BIT DEFAULT 0,
    AllowOnlineBooking BIT DEFAULT 1,
    MaxAdvanceBookingDays INT DEFAULT 30,
    MinAdvanceBookingHours INT DEFAULT 2,
    IsGroupService BIT DEFAULT 0, -- Para servicios grupales
    MaxGroupSize INT DEFAULT 1,
    IsMobileService BIT DEFAULT 0, -- Servicio a domicilio
    DisplayOrder INT DEFAULT 0,
    CreatedAt DATETIME2 DEFAULT GETUTCDATE(),
    UpdatedAt DATETIME2 DEFAULT GETUTCDATE(),
    FOREIGN KEY (BusinessId) REFERENCES Businesses(BusinessId) ON DELETE CASCADE,
    FOREIGN KEY (CategoryId) REFERENCES ServiceCategories(CategoryId),
    INDEX IX_Services_BusinessId (BusinessId),
    INDEX IX_Services_CategoryId (CategoryId)
);

-- Servicios que pueden realizar cada empleado
CREATE TABLE EmployeeServices (
    EmployeeServiceId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    EmployeeId UNIQUEIDENTIFIER NOT NULL,
    ServiceId UNIQUEIDENTIFIER NOT NULL,
    FOREIGN KEY (EmployeeId) REFERENCES BusinessEmployees(EmployeeId) ON DELETE CASCADE,
    FOREIGN KEY (ServiceId) REFERENCES Services(ServiceId) ON DELETE CASCADE,
    UNIQUE (EmployeeId, ServiceId),
    INDEX IX_EmployeeServices_EmployeeId (EmployeeId),
    INDEX IX_EmployeeServices_ServiceId (ServiceId)
);

-- Paquetes de servicios
CREATE TABLE ServicePackages (
    PackageId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    BusinessId UNIQUEIDENTIFIER NOT NULL,
    PackageName NVARCHAR(200) NOT NULL,
    Description NVARCHAR(1000),
    TotalPrice DECIMAL(10,2) NOT NULL,
    DiscountPercentage DECIMAL(5,2),
    ImageUrl NVARCHAR(500),

    CreatedAt DATETIME2 DEFAULT GETUTCDATE(),
    FOREIGN KEY (BusinessId) REFERENCES Businesses(BusinessId) ON DELETE CASCADE,
    INDEX IX_ServicePackages_BusinessId (BusinessId)
);

CREATE TABLE PackageServices (
    PackageServiceId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    PackageId UNIQUEIDENTIFIER NOT NULL,
    ServiceId UNIQUEIDENTIFIER NOT NULL,
    ServiceOrder INT DEFAULT 0, -- Orden en secuencia
    FOREIGN KEY (PackageId) REFERENCES ServicePackages(PackageId) ON DELETE CASCADE,
    FOREIGN KEY (ServiceId) REFERENCES Services(ServiceId),
    INDEX IX_PackageServices_PackageId (PackageId)

CREATE TABLE Appointments (
    AppointmentId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    BusinessId UNIQUEIDENTIFIER NOT NULL,
    LocationId UNIQUEIDENTIFIER NOT NULL,
    ClientId UNIQUEIDENTIFIER NOT NULL,
    EmployeeId UNIQUEIDENTIFIER, -- NULL si no hay preferencia
    ServiceId UNIQUEIDENTIFIER,
    PackageId UNIQUEIDENTIFIER, -- NULL si no es paquete
    StatusId INT NOT NULL DEFAULT 1,
    AppointmentDate DATE NOT NULL,
    StartTime TIME NOT NULL,
    EndTime TIME NOT NULL,
    DurationMinutes INT NOT NULL,
    Notes NVARCHAR(1000), -- Notas del cliente
    InternalNotes NVARCHAR(1000), -- Notas internas del negocio
    ClientAddress NVARCHAR(500), -- Para servicios a domicilio
    TotalPrice DECIMAL(10,2),
    IsRecurring BIT DEFAULT 0,
    RecurrencePattern NVARCHAR(MAX), -- JSON con patrón de recurrencia
    ParentAppointmentId UNIQUEIDENTIFIER, -- Para citas recurrentes
    RequiresConfirmation BIT DEFAULT 0,
    ConfirmedAt DATETIME2,
    CancelledAt DATETIME2,
    CancellationReason NVARCHAR(500),
    CheckedInAt DATETIME2,
    CompletedAt DATETIME2,
    CreatedAt DATETIME2 DEFAULT GETUTCDATE(),
    UpdatedAt DATETIME2 DEFAULT GETUTCDATE(),
    FOREIGN KEY (BusinessId) REFERENCES Businesses(BusinessId),
    FOREIGN KEY (LocationId) REFERENCES BusinessLocations(LocationId),
    FOREIGN KEY (ClientId) REFERENCES Users(UserId),
    FOREIGN KEY (EmployeeId) REFERENCES BusinessEmployees(EmployeeId),
    FOREIGN KEY (ServiceId) REFERENCES Services(ServiceId),
    FOREIGN KEY (PackageId) REFERENCES ServicePackages(PackageId),
    FOREIGN KEY (StatusId) REFERENCES AppointmentStatuses(StatusId),
    FOREIGN KEY (ParentAppointmentId) REFERENCES Appointments(AppointmentId),
    INDEX IX_Appointments_BusinessId (BusinessId),
    INDEX IX_Appointments_ClientId (ClientId),
    INDEX IX_Appointments_EmployeeId (EmployeeId),
    INDEX IX_Appointments_Date (AppointmentDate),
    INDEX IX_Appointments_Status (StatusId)
);

-- Para citas grupales (múltiples clientes)
CREATE TABLE AppointmentParticipants (
    ParticipantId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    AppointmentId UNIQUEIDENTIFIER NOT NULL,
    ClientId UNIQUEIDENTIFIER NOT NULL,
    Status NVARCHAR(20) DEFAULT 'confirmed',
    JoinedAt DATETIME2 DEFAULT GETUTCDATE(),
    FOREIGN KEY (AppointmentId) REFERENCES Appointments(AppointmentId) ON DELETE CASCADE,
    FOREIGN KEY (ClientId) REFERENCES Users(UserId),
    INDEX IX_AppointmentParticipants_AppointmentId (AppointmentId)
);

-- Recursos asignados a una cita
CREATE TABLE AppointmentResources (
    AppointmentResourceId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    AppointmentId UNIQUEIDENTIFIER NOT NULL,
    ResourceId UNIQUEIDENTIFIER NOT NULL,
    FOREIGN KEY (AppointmentId) REFERENCES Appointments(AppointmentId) ON DELETE CASCADE,
    FOREIGN KEY (ResourceId) REFERENCES Resources(ResourceId),
    INDEX IX_AppointmentResources_AppointmentId (AppointmentId)
);

-- =============================================
-- TABLAS DE RESEÑAS Y RATINGS
-- =============================================

CREATE TABLE Reviews (
    ReviewId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    AppointmentId UNIQUEIDENTIFIER NOT NULL,
    BusinessId UNIQUEIDENTIFIER NOT NULL,
    ClientId UNIQUEIDENTIFIER NOT NULL,
    EmployeeId UNIQUEIDENTIFIER,
    Rating INT NOT NULL CHECK (Rating >= 1 AND Rating <= 5),
    Comment NVARCHAR(1000),
    IsPublic BIT DEFAULT 1,
    CreatedAt DATETIME2 DEFAULT GETUTCDATE(),
    FOREIGN KEY (AppointmentId) REFERENCES Appointments(AppointmentId),
    FOREIGN KEY (BusinessId) REFERENCES Businesses(BusinessId),
    FOREIGN KEY (ClientId) REFERENCES Users(UserId),
    FOREIGN KEY (EmployeeId) REFERENCES BusinessEmployees(EmployeeId),
    UNIQUE (AppointmentId), -- Solo una reseña por cita
    INDEX IX_Reviews_BusinessId (BusinessId),
    INDEX IX_Reviews_EmployeeId (EmployeeId)
);

-- TABLAS DE FACTURACIÓN Y PAGOS

CREATE TABLE Invoices (
    InvoiceId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    BusinessId UNIQUEIDENTIFIER NOT NULL,
    InvoiceNumber NVARCHAR(50) NOT NULL UNIQUE,
    BillingPeriodStart DATE NOT NULL,
    BillingPeriodEnd DATE NOT NULL,
    TotalAppointments INT NOT NULL,
    BasePrice DECIMAL(10,2) NOT NULL,
    AppointmentCharges DECIMAL(10,2) NOT NULL,
    Subtotal DECIMAL(10,2) NOT NULL,
    TaxPercentage DECIMAL(5,2) DEFAULT 13.00, -- IVA en Costa Rica
    TaxAmount DECIMAL(10,2) NOT NULL,
    Total DECIMAL(10,2) NOT NULL,
    Currency NVARCHAR(3) DEFAULT 'USD',
    Status NVARCHAR(20) DEFAULT 'pending', -- pending, paid, overdue, cancelled
    DueDate DATE NOT NULL,
    PaidAt DATETIME2,
    ElectronicInvoiceKey NVARCHAR(100), -- Clave numérica Hacienda CR
    ElectronicInvoiceXml NVARCHAR(MAX), -- XML de factura electrónica
    CreatedAt DATETIME2 DEFAULT GETUTCDATE(),
    FOREIGN KEY (BusinessId) REFERENCES Businesses(BusinessId),
    INDEX IX_Invoices_BusinessId (BusinessId),
    INDEX IX_Invoices_Status (Status)
);

CREATE TABLE Payments (
    PaymentId UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
    InvoiceId UNIQUEIDENTIFIER NOT NULL,
    StripePaymentIntentId NVARCHAR(255),
    Amount DECIMAL(10,2) NOT NULL,
    Currency NVARCHAR(3) DEFAULT 'USD',
    Status NVARCHAR(20) DEFAULT 'pending', -- pending, succeeded, failed, refunded
    PaymentMethod NVARCHAR(50), -- card, bank_transfer, etc.
    FailureReason NVARCHAR(500),
    ProcessedAt DATETIME2,
    CreatedAt DATETIME2 DEFAULT GETUTCDATE(),
    FOREIGN KEY (InvoiceId) REFERENCES Invoices(InvoiceId),
    INDEX IX_Payments_InvoiceId (InvoiceId),
    INDEX IX_Payments_StripePaymentIntentId (StripePaymentIntentId)
);

CREATE TABLE Coupons (

-- SP: Obtener disponibilidad de un empleado en una fecha
CREATE PROCEDURE sp_GetEmployeeAvailability
    @EmployeeId UNIQUEIDENTIFIER,
    @Date DATE,
    @ServiceId UNIQUEIDENTIFIER
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @ServiceDuration INT;
    DECLARE @DayOfWeek INT = DATEPART(WEEKDAY, @Date);

    -- Obtener duración del servicio
    SELECT @ServiceDuration = DurationMinutes 
    FROM Services 
    WHERE ServiceId = @ServiceId;

    -- Verificar si el empleado trabaja ese día
    SELECT 
        ewh.StartTime,
        ewh.EndTime,
        @ServiceDuration as ServiceDuration
    FROM EmployeeWorkingHours ewh
    WHERE ewh.EmployeeId = @EmployeeId 
        AND ewh.DayOfWeek = @DayOfWeek
        AND ewh.IsAvailable = 1
        AND NOT EXISTS (
            -- Verificar si el empleado tiene tiempo libre ese día
            SELECT 1 
            FROM EmployeeTimeOff eto
            WHERE eto.EmployeeId = @EmployeeId
                AND @Date BETWEEN CAST(eto.StartDate as DATE) AND CAST(eto.EndDate as DATE)
        );

    -- Obtener citas existentes del empleado ese día
    SELECT 
        StartTime,
        EndTime,
        DurationMinutes,
        s.ServiceName
    FROM Appointments a
    LEFT JOIN Services s ON a.ServiceId = s.ServiceId
    WHERE a.EmployeeId = @EmployeeId 
        AND a.AppointmentDate = @Date
        AND a.StatusId NOT IN (5, 6, 7); -- Excluir canceladas y no-show
END;
GO

-- SP: Crear una cita con validaciones
CREATE PROCEDURE sp_CreateAppointment
    @BusinessId UNIQUEIDENTIFIER,
    @LocationId UNIQUEIDENTIFIER,
    @ClientId UNIQUEIDENTIFIER,
    @EmployeeId UNIQUEIDENTIFIER = NULL,
    @ServiceId UNIQUEIDENTIFIER = NULL,
    @PackageId UNIQUEIDENTIFIER = NULL,
    @AppointmentDate DATE,
    @StartTime TIME,
    @Notes NVARCHAR(1000) = NULL,
    @ClientAddress NVARCHAR(500) = NULL,
    @NewAppointmentId UNIQUEIDENTIFIER OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION;

        DECLARE @DurationMinutes INT;
        DECLARE @TotalPrice DECIMAL(10,2);
        DECLARE @EndTime TIME;
        DECLARE @RequiresConfirmation BIT;
        DECLARE @IsMobileService BIT = 0;

        -- Validar que el servicio o paquete exista
        IF @ServiceId IS NOT NULL
        BEGIN
            SELECT 
                @DurationMinutes = DurationMinutes,
                @TotalPrice = Price,
                @IsMobileService = IsMobileService
            FROM Services 
            WHERE ServiceId = @ServiceId AND BusinessId = @BusinessId;

            IF @DurationMinutes IS NULL
            BEGIN
                RAISERROR('Servicio no encontrado', 16, 1);
                RETURN;
            END
        END
        ELSE IF @PackageId IS NOT NULL
        BEGIN
            SELECT 
                @TotalPrice = TotalPrice
            FROM ServicePackages 
            WHERE PackageId = @PackageId AND BusinessId = @BusinessId;

            -- Calcular duración total del paquete
            SELECT @DurationMinutes = SUM(s.DurationMinutes)
            FROM PackageServices ps
            INNER JOIN Services s ON ps.ServiceId = s.ServiceId
            WHERE ps.PackageId = @PackageId;
        END
        ELSE
        BEGIN
            RAISERROR('Debe especificar un servicio o paquete', 16, 1);
            RETURN;
        END

        -- Validar dirección para servicios móviles
        IF @IsMobileService = 1 AND @ClientAddress IS NULL
        BEGIN
            RAISERROR('Debe proporcionar dirección para servicios a domicilio', 16, 1);
            RETURN;
        END

        -- Calcular hora de fin
        SET @EndTime = DATEADD(MINUTE, @DurationMinutes, CAST(@StartTime as DATETIME));

        -- Obtener configuración de confirmación
        SELECT @RequiresConfirmation = CAST(SettingValue as BIT)
        FROM BusinessSettings
        WHERE BusinessId = @BusinessId 
            AND SettingKey = 'require_appointment_confirmation';

        SET @RequiresConfirmation = ISNULL(@RequiresConfirmation, 0);

        -- Validar que el empleado esté disponible (si se especificó)
        IF @EmployeeId IS NOT NULL
        BEGIN
            IF EXISTS (
                SELECT 1 
                FROM Appointments 
                WHERE EmployeeId = @EmployeeId 
                    AND AppointmentDate = @AppointmentDate
                    AND StatusId NOT IN (5, 6, 7)
                    AND (
                        (@StartTime >= StartTime AND @StartTime < EndTime)
                        OR (@EndTime > StartTime AND @EndTime <= EndTime)
                        OR (@StartTime <= StartTime AND @EndTime >= EndTime)
                    )
            )
            BEGIN
                RAISERROR('El empleado no está disponible en ese horario', 16, 1);
                RETURN;
            END
        END

        -- Crear la cita
        SET @NewAppointmentId = NEWID();

        INSERT INTO Appointments (
            AppointmentId, BusinessId, LocationId, ClientId, EmployeeId, 
            ServiceId, PackageId, StatusId, AppointmentDate, StartTime, 
            EndTime, DurationMinutes, Notes, ClientAddress, TotalPrice, 
            RequiresConfirmation
        )
        VALUES (
            @NewAppointmentId, @BusinessId, @LocationId, @ClientId, @EmployeeId,
            @ServiceId, @PackageId, 1, @AppointmentDate, @StartTime,
            @EndTime, @DurationMinutes, @Notes, @ClientAddress, @TotalPrice,
            @RequiresConfirmation
        );

        COMMIT TRANSACTION;

        -- Devolver la cita creada
        SELECT * FROM Appointments WHERE AppointmentId = @NewAppointmentId;

    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        
        DECLARE @ErrorMessage NVARCHAR(4000) = ERROR_MESSAGE();
        DECLARE @ErrorSeverity INT = ERROR_SEVERITY();
        DECLARE @ErrorState INT = ERROR_STATE();
        
        RAISERROR(@ErrorMessage, @ErrorSeverity, @ErrorState);
    END CATCH
END;
GO

-- SP: Cancelar cita con validación de ventana de cancelación
CREATE PROCEDURE sp_CancelAppointment
    @AppointmentId UNIQUEIDENTIFIER,
    @CancelledBy UNIQUEIDENTIFIER, -- UserId que cancela
    @CancellationReason NVARCHAR(500) = NULL,
    @Success BIT OUTPUT,
    @Message NVARCHAR(500) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION;

        DECLARE @BusinessId UNIQUEIDENTIFIER;
        DECLARE @AppointmentDate DATE;
        DECLARE @StartTime TIME;
        DECLARE @CancellationWindowHours INT;
        DECLARE @AppointmentDateTime DATETIME2;
        DECLARE @HoursUntilAppointment FLOAT;

        -- Obtener información de la cita
        SELECT 
            @BusinessId = BusinessId,
            @AppointmentDate = AppointmentDate,
            @StartTime = StartTime,
            @AppointmentDateTime = CAST(CAST(AppointmentDate as DATETIME) + CAST(StartTime as DATETIME) as DATETIME2)
        FROM Appointments
        WHERE AppointmentId = @AppointmentId;

        -- Obtener ventana de cancelación del negocio
        SELECT @CancellationWindowHours = CAST(SettingValue as INT)
        FROM BusinessSettings
        WHERE BusinessId = @BusinessId 
            AND SettingKey = 'cancellation_window_hours';

        SET @CancellationWindowHours = ISNULL(@CancellationWindowHours, 24);

        -- Calcular horas hasta la cita
        SET @HoursUntilAppointment = DATEDIFF(HOUR, GETUTCDATE(), @AppointmentDateTime);

        -- Validar ventana de cancelación
        IF @HoursUntilAppointment < @CancellationWindowHours
        BEGIN
            SET @Success = 0;
            SET @Message = 'No se puede cancelar. Debe cancelar con al menos ' + 
                          CAST(@CancellationWindowHours as NVARCHAR) + ' horas de anticipación.';
            ROLLBACK TRANSACTION;
            RETURN;
        END

        -- Cancelar la cita
        UPDATE Appointments
        SET 
            StatusId = 5, -- cancelled_by_client
            CancelledAt = GETUTCDATE(),
            CancellationReason = @CancellationReason,
            UpdatedAt = GETUTCDATE()
        WHERE AppointmentId = @AppointmentId;

        SET @Success = 1;
        SET @Message = 'Cita cancelada exitosamente';

        COMMIT TRANSACTION;

    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        
        SET @Success = 0;
        SET @Message = ERROR_MESSAGE();
    END CATCH
END;
GO

-- SP: Generar factura mensual para un negocio
CREATE PROCEDURE sp_GenerateMonthlyInvoice
    @BusinessId UNIQUEIDENTIFIER,
    @BillingMonth DATE -- Primer día del mes a facturar
AS
BEGIN
    SET NOCOUNT ON;
    BEGIN TRY
        BEGIN TRANSACTION;

        DECLARE @BillingPeriodStart DATE = @BillingMonth;
        DECLARE @BillingPeriodEnd DATE = EOMONTH(@BillingMonth);
        DECLARE @IndustryId INT;
        DECLARE @BasePriceMonthly DECIMAL(10,2);
        DECLARE @PricePerAppointment DECIMAL(10,2);
        DECLARE @TotalAppointments INT;
        DECLARE @FreeAppointments INT = 50; -- Primeras 50 incluidas en base
        DECLARE @AppointmentCharges DECIMAL(10,2);
        DECLARE @Subtotal DECIMAL(10,2);
        DECLARE @TaxPercentage DECIMAL(5,2) = 13.00; -- IVA CR
        DECLARE @TaxAmount DECIMAL(10,2);
        DECLARE @Total DECIMAL(10,2);
        DECLARE @InvoiceNumber NVARCHAR(50);
        DECLARE @NewInvoiceId UNIQUEIDENTIFIER = NEWID();

        -- Obtener información del negocio
        SELECT 
            @IndustryId = b.IndustryId,
            @BasePriceMonthly = i.BasePriceMonthly,
            @PricePerAppointment = i.PricePerAppointment
        FROM Businesses b
        INNER JOIN Industries i ON b.IndustryId = i.IndustryId
        WHERE b.BusinessId = @BusinessId;

        -- Contar citas completadas en el período
        SELECT @TotalAppointments = COUNT(*)
        FROM Appointments
        WHERE BusinessId = @BusinessId
            AND AppointmentDate BETWEEN @BillingPeriodStart AND @BillingPeriodEnd
            AND StatusId = 4; -- completed

        -- Calcular cargos por citas adicionales
        IF @TotalAppointments > @FreeAppointments
            SET @AppointmentCharges = (@TotalAppointments - @FreeAppointments) * @PricePerAppointment;
        ELSE
            SET @AppointmentCharges = 0;

        -- Calcular totales
        SET @Subtotal = @BasePriceMonthly + @AppointmentCharges;
        SET @TaxAmount = @Subtotal * (@TaxPercentage / 100);
        SET @Total = @Subtotal + @TaxAmount;

        -- Generar número de factura
        SET @InvoiceNumber = 'INV-' + 
                           FORMAT(@BillingMonth, 'yyyyMM') + '-' + 
                           RIGHT('00000' + CAST(ABS(CHECKSUM(@BusinessId)) % 100000 as NVARCHAR), 5);

        -- Crear factura
        INSERT INTO Invoices (
            InvoiceId, BusinessId, InvoiceNumber, BillingPeriodStart, BillingPeriodEnd,
            TotalAppointments, BasePrice, AppointmentCharges, Subtotal,
            TaxPercentage, TaxAmount, Total, DueDate
        )
        VALUES (
            @NewInvoiceId, @BusinessId, @InvoiceNumber, @BillingPeriodStart, @BillingPeriodEnd,
            @TotalAppointments, @BasePriceMonthly, @AppointmentCharges, @Subtotal,
            @TaxPercentage, @TaxAmount, @Total, 
            DATEADD(DAY, 15, @BillingPeriodEnd) -- 15 días para pagar
        );

        COMMIT TRANSACTION;

        -- Devolver la factura generada
        SELECT * FROM Invoices WHERE InvoiceId = @NewInvoiceId;

    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0
            ROLLBACK TRANSACTION;
        
        THROW;
    END CATCH
END;
GO

-- SP: Obtener slots disponibles para un servicio
CREATE PROCEDURE sp_GetAvailableSlots
    @BusinessId UNIQUEIDENTIFIER,
    @LocationId UNIQUEIDENTIFIER,
    @ServiceId UNIQUEIDENTIFIER,
    @Date DATE,
    @EmployeeId UNIQUEIDENTIFIER = NULL -- NULL = cualquier empleado
AS
BEGIN
    SET NOCOUNT ON;

    DECLARE @DayOfWeek INT = DATEPART(WEEKDAY, @Date);
    DECLARE @ServiceDuration INT;
    DECLARE @BufferTime INT;

    -- Obtener duración del servicio
    SELECT @ServiceDuration = DurationMinutes FROM Services WHERE ServiceId = @ServiceId;

    -- Obtener tiempo de buffer entre citas
    SELECT @BufferTime = CAST(SettingValue as INT)
    FROM BusinessSettings
    WHERE BusinessId = @BusinessId AND SettingKey = 'buffer_time_minutes';

    SET @BufferTime = ISNULL(@BufferTime, 0);

    -- Si se especificó empleado, obtener solo sus slots
    IF @EmployeeId IS NOT NULL
    BEGIN
        SELECT 
            @EmployeeId as EmployeeId,
            be.DisplayName as EmployeeName,
            ewh.StartTime as ShiftStart,
            ewh.EndTime as ShiftEnd,
            @ServiceDuration as ServiceDuration,
            @BufferTime as BufferTime
        FROM EmployeeWorkingHours ewh
        INNER JOIN BusinessEmployees be ON ewh.EmployeeId = be.EmployeeId
        WHERE ewh.EmployeeId = @EmployeeId
            AND ewh.DayOfWeek = @DayOfWeek
            AND ewh.IsAvailable = 1
            AND be.IsActive = 1
            AND NOT EXISTS (
                SELECT 1 FROM EmployeeTimeOff eto
                WHERE eto.EmployeeId = @EmployeeId
                    AND @Date BETWEEN CAST(eto.StartDate as DATE) AND CAST(eto.EndDate as DATE)
            );
    END
    ELSE
    BEGIN
        -- Obtener todos los empleados que pueden hacer ese servicio
        SELECT 
            be.EmployeeId,
            be.DisplayName as EmployeeName,
            ewh.StartTime as ShiftStart,
            ewh.EndTime as ShiftEnd,
            @ServiceDuration as ServiceDuration,
            @BufferTime as BufferTime
        FROM BusinessEmployees be
        INNER JOIN EmployeeServices es ON be.EmployeeId = es.EmployeeId
        INNER JOIN EmployeeWorkingHours ewh ON be.EmployeeId = ewh.EmployeeId
        WHERE es.ServiceId = @ServiceId
            AND be.BusinessId = @BusinessId
            AND be.IsActive = 1
            AND ewh.DayOfWeek = @DayOfWeek
            AND ewh.IsAvailable = 1
            AND NOT EXISTS (
                SELECT 1 FROM EmployeeTimeOff eto
                WHERE eto.EmployeeId = be.EmployeeId
                    AND @Date BETWEEN CAST(eto.StartDate as DATE) AND CAST(eto.EndDate as DATE)
            );
    END

    -- Obtener citas existentes para ese día
    SELECT 
        EmployeeId,
        StartTime,
        EndTime,
        DurationMinutes
    FROM Appointments
    WHERE LocationId = @LocationId
        AND AppointmentDate = @Date
        AND StatusId NOT IN (5, 6, 7)
        AND (@EmployeeId IS NULL OR EmployeeId = @EmployeeId);
END;
GO

-- Función para calcular rating promedio de un negocio
CREATE FUNCTION fn_GetBusinessAverageRating(@BusinessId UNIQUEIDENTIFIER)
RETURNS DECIMAL(3,2)
AS
BEGIN
    DECLARE @AvgRating DECIMAL(3,2);
    
    SELECT @AvgRating = AVG(CAST(Rating as DECIMAL(3,2)))
    FROM Reviews
    WHERE BusinessId = @BusinessId AND IsPublic = 1;
    
    RETURN ISNULL(@AvgRating, 0);
END;
GO

-- Función para verificar si un negocio está activo y pagado
CREATE FUNCTION fn_IsBusinessActive(@BusinessId UNIQUEIDENTIFIER)
RETURNS BIT
AS
BEGIN
    DECLARE @IsActive BIT = 0;
    
    SELECT @IsActive = CASE 
        WHEN SubscriptionStatus IN ('trial', 'active') 
            AND (TrialEndsAt > GETUTCDATE() OR SubscriptionEndDate > GETUTCDATE())
        THEN 1
        ELSE 0
    END
    FROM Businesses
    WHERE BusinessId = @BusinessId;
    
    RETURN ISNULL(@IsActive, 0);
END;
GO

-- Función para obtener el próximo slot disponible
CREATE FUNCTION fn_GetNextAvailableSlot(
    @EmployeeId UNIQUEIDENTIFIER,
    @Date DATE,
    @DurationMinutes INT
)
RETURNS TIME
AS
BEGIN
    DECLARE @NextSlot TIME;
    
    -- Esta es una función simplificada
    -- En producción, necesitarías lógica más compleja
    SELECT TOP 1 @NextSlot = DATEADD(MINUTE, DurationMinutes + 15, EndTime)
    FROM Appointments
    WHERE EmployeeId = @EmployeeId
        AND AppointmentDate = @Date
        AND StatusId NOT IN (5, 6, 7)
    ORDER BY EndTime DESC;
    
    RETURN @NextSlot;
END;
GO

-- Trigger para auditar cambios en citas
CREATE TRIGGER trg_Appointments_Audit
ON Appointments
AFTER UPDATE, DELETE
AS
BEGIN
    SET NOCOUNT ON;
    
    DECLARE @Action NVARCHAR(10);
    
    IF EXISTS (SELECT * FROM deleted) AND EXISTS (SELECT * FROM inserted)
        SET @Action = 'UPDATE';
    ELSE IF EXISTS (SELECT * FROM deleted)
        SET @Action = 'DELETE';
    
    INSERT INTO AuditLogs (
        Action, 
        EntityType, 
        EntityId, 
        OldValues, 
        NewValues
    )
    SELECT 
        @Action,
        'Appointment',
        ISNULL(i.AppointmentId, d.AppointmentId),
        (SELECT d.* FOR JSON PATH, WITHOUT_ARRAY_WRAPPER),
        (SELECT i.* FOR JSON PATH, WITHOUT_ARRAY_WRAPPER)
    FROM inserted i
    FULL OUTER JOIN deleted d ON i.AppointmentId = d.AppointmentId;
END;
GO

-- Índices compuestos para queries frecuentes
CREATE NONCLUSTERED INDEX IX_Appointments_Business_Date_Status
ON Appointments(BusinessId, AppointmentDate, StatusId)
INCLUDE (ClientId, EmployeeId, ServiceId, StartTime, EndTime);
GO

CREATE NONCLUSTERED INDEX IX_Appointments_Employee_Date
ON Appointments(EmployeeId, AppointmentDate)
INCLUDE (StartTime, EndTime, StatusId)
WHERE StatusId NOT IN (5, 6, 7);
GO

CREATE NONCLUSTERED INDEX IX_Reviews_Business_Rating
ON Reviews(BusinessId, Rating)
INCLUDE (EmployeeId, CreatedAt)
WHERE IsPublic = 1;
GO

CREATE NONCLUSTERED INDEX IX_Invoices_Business_Status_Period
ON Invoices(BusinessId, Status, BillingPeriodStart)
INCLUDE (Total, DueDate);
GO

-- Índices para búsqueda de texto
CREATE NONCLUSTERED INDEX IX_Businesses_Name
ON Businesses(BusinessName)
INCLUDE (IndustryId, IsActive, SubscriptionStatus);
GO

CREATE NONCLUSTERED INDEX IX_Services_Name
ON Services(ServiceName)
INCLUDE (BusinessId, Price, DurationMinutes, IsActive);
